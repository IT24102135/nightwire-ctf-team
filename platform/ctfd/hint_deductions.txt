#!/usr/bin/env python3
"""
NIGHTWIRE CTFd Hint Score Adjustment Service
Monitors hint views and automatically deducts points from player scores.
"""

import mysql.connector
import time
import sys
from datetime import datetime
from mysql.connector import Error as MySQLError

# Configuration
DB_CONFIG = {
    'host': 'localhost',
    'port': 3306,
    'user': 'ctfd',
    'password': 'ctfd',
    'database': 'ctfd'
}

HINT_COST = 5  # Points deducted per hint

# Track processed unlocks to avoid double-deducting
processed_unlocks = set()

def log(message):
    """Log with timestamp"""
    timestamp = datetime.now().strftime('%Y-%m-%d %H:%M:%S.%f')
    print(f"[{timestamp}] {message}")
    sys.stdout.flush()

def get_db_connection():
    """Create database connection"""
    try:
        conn = mysql.connector.connect(**DB_CONFIG)
        return conn
    except MySQLError as e:
        log(f"Database connection error: {e}")
        return None

def create_deductions_table():
    """Create hint_deductions table if it doesn't exist"""
    conn = get_db_connection()
    if not conn:
        return False
    
    try:
        cursor = conn.cursor()
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS hint_deductions (
                id INT AUTO_INCREMENT PRIMARY KEY,
                user_id INT NOT NULL,
                challenge_id INT NOT NULL,
                hint_id INT NOT NULL,
                cost INT NOT NULL,
                created DATETIME DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (user_id) REFERENCES users(id),
                FOREIGN KEY (challenge_id) REFERENCES challenges(id)
            )
        """)
        conn.commit()
        cursor.close()
        conn.close()
        return True
    except MySQLError as e:
        log(f"Error creating table: {e}")
        return False

def deduct_hint_cost(user_id, challenge_id, hint_id, cost):
    """Record hint deduction in hint_deductions table"""
    conn = get_db_connection()
    if not conn:
        return False
    
    try:
        cursor = conn.cursor()
        
        # Insert deduction record
        cursor.execute("""
            INSERT INTO hint_deductions (user_id, challenge_id, hint_id, cost)
            VALUES (%s, %s, %s, %s)
        """, (user_id, challenge_id, hint_id, cost))
        conn.commit()
        
        # Get current total score
        cursor.execute("""
            SELECT COALESCE(SUM(value), 0) as solve_score
            FROM solves
            WHERE user_id = %s AND solved = 1
        """, (user_id,))
        solve_score = cursor.fetchone()[0]
        
        # Get total deductions
        cursor.execute("""
            SELECT COALESCE(SUM(cost), 0) as total_deductions
            FROM hint_deductions
            WHERE user_id = %s
        """, (user_id,))
        total_deductions = cursor.fetchone()[0]
        
        final_score = max(0, solve_score - total_deductions)
        
        log(f"✓ User {user_id} - Challenge {challenge_id}: Solves={solve_score}, Deductions={total_deductions}, Final={final_score}")
        
        cursor.close()
        conn.close()
        return True
        
    except MySQLError as e:
        log(f"✗ Database error for user {user_id}: {e}")
        return False

def process_hint_views():
    """Check for new hint views and record deductions"""
    conn = get_db_connection()
    if not conn:
        return
    
    try:
        cursor = conn.cursor(dictionary=True)
        
        # Query all hint unlocks
        query = """
        SELECT u.id, u.user_id, h.id as hint_id, h.challenge_id, h.cost
        FROM unlocks u
        JOIN hints h ON u.target = h.id
        WHERE u.type = 'hints'
        ORDER BY u.id
        """
        
        cursor.execute(query)
        results = cursor.fetchall()
        
        new_unlocks = []
        for row in results:
            unlock_id = row['id']
            if unlock_id not in processed_unlocks:
                new_unlocks.append(row)
                processed_unlocks.add(unlock_id)
        
        if new_unlocks:
            log(f"Found {len(new_unlocks)} new hint view(s)")
            for row in new_unlocks:
                user_id = row['user_id']
                challenge_id = row['challenge_id']
                hint_id = row['hint_id']
                cost = row['cost'] if row['cost'] else HINT_COST
                
                deduct_hint_cost(user_id, challenge_id, hint_id, cost)
        
        cursor.close()
        
    except MySQLError as e:
        log(f"Database query error: {e}")
    finally:
        conn.close()

def main():
    """Main service loop"""
    log("=" * 60)
    log("NIGHTWIRE Hint Score Adjustment Service started")
    log("Creating hint_deductions table...")
    
    if create_deductions_table():
        log("✓ hint_deductions table ready")
    else:
        log("✗ Failed to create table, continuing anyway...")
    
    log("Monitoring for hint views... (checking every 5 seconds)")
    log("Press Ctrl+C to stop")
    log("=" * 60)
    
    try:
        while True:
            process_hint_views()
            time.sleep(5)
    except KeyboardInterrupt:
        log("\nService stopped by user")
        sys.exit(0)
    except Exception as e:
        log(f"Fatal error: {e}")
        sys.exit(1)

if __name__ == '__main__':
    main()
